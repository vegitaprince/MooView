cmake_minimum_required(VERSION 3.24)
project(MooViewExr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(OPENEXR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OPENEXR_INSTALL OFF CACHE BOOL "" FORCE)
set(IMATH_INSTALL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(openexr GIT_REPOSITORY https://github.com/AcademySoftwareFoundation/openexr.git GIT_TAG 267838d6a4719cda765081da9c22901d3651de83 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(openexr)
add_library(MooViewExr SHARED MooViewExr.cpp)
target_link_libraries(MooViewExr PRIVATE OpenEXR::OpenEXR)
add_executable(ExrFixtures ExrFixtures.cpp)
target_link_libraries(ExrFixtures PRIVATE OpenEXR::OpenEXR)
